-- ============================================
-- HAAT BARAN - DONATION MANAGEMENT SYSTEM - POSTGRESQL SCHEMA
-- ============================================


-- ============================================
-- 1. CORE USER TABLES
-- ============================================

-- Base user table (abstract User class in UML)
CREATE TABLE users (
    user_id SERIAL PRIMARY KEY,
    username VARCHAR(50) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    phone VARCHAR(20),
    user_type VARCHAR(20) NOT NULL CHECK (user_type IN ('VOLUNTEER', 'DONOR', 'ADMINISTRATOR')),
    account_status VARCHAR(20) DEFAULT 'ACTIVE' CHECK (account_status IN ('ACTIVE', 'INACTIVE', 'SUSPENDED', 'PENDING')),
    credentials TEXT, -- Could store JWT tokens or auth credentials
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    last_login_at TIMESTAMP,
    
    -- Indexes for performance
    INDEX idx_user_type (user_type),
    INDEX idx_account_status (account_status),
    INDEX idx_email (email)
);

-- Volunteer-specific table (extends users)
CREATE TABLE volunteers (
    volunteer_id SERIAL PRIMARY KEY,
    user_id INTEGER UNIQUE NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
    dashboard_settings JSONB DEFAULT '{}'::jsonb,
    assigned_region VARCHAR(100),
    performance_score DECIMAL(5,2) DEFAULT 0.00,
    total_applications_collected INTEGER DEFAULT 0,
    last_active_at TIMESTAMP,
    metadata JSONB DEFAULT '{}'::jsonb,
    
    INDEX idx_volunteer_region (assigned_region)
);

-- Donor-specific table (extends users)
CREATE TABLE donors (
    donor_id SERIAL PRIMARY KEY,
    user_id INTEGER UNIQUE NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
    contact_details JSONB DEFAULT '{}'::jsonb, -- Could include address, secondary contacts
    payment_details JSONB DEFAULT '{}'::jsonb, -- Payment methods, billing info
    total_donated DECIMAL(15,2) DEFAULT 0.00,
    donation_count INTEGER DEFAULT 0,
    preferred_cause VARCHAR(100),
    donor_type VARCHAR(20) DEFAULT 'INDIVIDUAL' CHECK (donor_type IN ('INDIVIDUAL', 'CORPORATE', 'NGO', 'GOVERNMENT')),
    tax_id VARCHAR(50), -- For receipt/tax purposes
    metadata JSONB DEFAULT '{}'::jsonb,
    
    INDEX idx_donor_type (donor_type),
    INDEX idx_total_donated (total_donated)
);

-- Administrator-specific table (extends users)
CREATE TABLE administrators (
    admin_id SERIAL PRIMARY KEY,
    user_id INTEGER UNIQUE NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
    web_dashboard_settings JSONB DEFAULT '{}'::jsonb,
    access_level VARCHAR(20) DEFAULT 'MODERATOR' CHECK (access_level IN ('VIEWER', 'MODERATOR', 'SUPER_ADMIN')),
    department VARCHAR(50),
    permissions JSONB DEFAULT '[]'::jsonb, -- Array of permission strings
    last_report_generated_at TIMESTAMP,
    metadata JSONB DEFAULT '{}'::jsonb
);

-- ============================================
-- 2. APPLICANT AND PROFILE TABLES
-- ============================================

-- Applicants table (not inheriting from users as per UML)
CREATE TABLE applicants (
    applicant_id SERIAL PRIMARY KEY,
    volunteer_id INTEGER REFERENCES volunteers(volunteer_id) ON DELETE SET NULL,
    admin_verifier_id INTEGER REFERENCES administrators(admin_id) ON DELETE SET NULL,
    
    -- Personal details (could be normalized further)
    personal_details JSONB DEFAULT '{}'::jsonb, -- name, dob, gender, etc.
    family_details JSONB DEFAULT '{}'::jsonb, -- family members, dependents
    business_details JSONB DEFAULT '{}'::jsonb, -- business type, plan, etc.
    
    -- Financial information
    funding_goal DECIMAL(15,2) NOT NULL CHECK (funding_goal > 0),
    funding_progress DECIMAL(15,2) DEFAULT 0.00 CHECK (funding_progress >= 0),
    financial_status VARCHAR(20) DEFAULT 'PENDING' CHECK (financial_status IN ('PENDING', 'REVIEWING', 'APPROVED', 'REJECTED')),
    
    -- Application tracking
    application_status VARCHAR(20) DEFAULT 'DRAFT' CHECK (application_status IN ('DRAFT', 'SUBMITTED', 'UNDER_REVIEW', 'APPROVED', 'REJECTED', 'FUNDED', 'COMPLETED')),
    submission_date TIMESTAMP,
    verification_date TIMESTAMP,
    approval_date TIMESTAMP,
    
    -- Additional metadata
    urgency_level VARCHAR(10) DEFAULT 'MEDIUM' CHECK (urgency_level IN ('LOW', 'MEDIUM', 'HIGH', 'CRITICAL')),
    category VARCHAR(50), -- Health, Education, Business, etc.
    tags TEXT[], -- Array of tags for filtering
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    -- Constraints
    CONSTRAINT chk_funding_progress CHECK (funding_progress <= funding_goal),
    
    -- Indexes
    INDEX idx_applicant_status (application_status),
    INDEX idx_financial_status (financial_status),
    INDEX idx_applicant_volunteer (volunteer_id),
    INDEX idx_applicant_admin (admin_verifier_id),
    INDEX idx_funding_goal (funding_goal),
    INDEX idx_created_at (created_at)
);

-- Applicant Profile table (composition relationship)
CREATE TABLE applicant_profiles (
    profile_id SERIAL PRIMARY KEY,
    applicant_id INTEGER UNIQUE NOT NULL REFERENCES applicants(applicant_id) ON DELETE CASCADE,
    
    -- Biometric data
    biometric_data JSONB DEFAULT '{}'::jsonb, -- Could store multiple biometric records
    fingerprint_hash VARCHAR(255), -- Hashed fingerprint data for security
    photograph_url TEXT, -- URL or path to stored photo
    photograph_verified BOOLEAN DEFAULT FALSE,
    
    -- Location data
    location_data JSONB DEFAULT '{}'::jsonb,
    gps_location POINT, -- PostgreSQL POINT type for geospatial queries
    latitude DECIMAL(10,8),
    longitude DECIMAL(11,8),
    upazilla VARCHAR(50),
    union_name VARCHAR(50), -- 'union' is a reserved word in SQL
    district VARCHAR(50),
    division VARCHAR(50),
    
    -- Verification tracking
    verification_status VARCHAR(20) DEFAULT 'PENDING' CHECK (verification_status IN ('PENDING', 'IN_PROGRESS', 'VERIFIED', 'FAILED')),
    verified_by INTEGER REFERENCES administrators(admin_id) ON DELETE SET NULL,
    verification_date TIMESTAMP,
    verification_notes TEXT,
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    -- Spatial index for location-based queries
    INDEX idx_gps_location (gps_location),
    INDEX idx_verification_status (verification_status),
    INDEX idx_location (division, district, upazilla, union_name)
);

-- ============================================
-- 3. DONATION AND TRANSACTION TABLES
-- ============================================

-- Donations table
CREATE TABLE donations (
    donation_id SERIAL PRIMARY KEY,
    donor_id INTEGER NOT NULL REFERENCES donors(donor_id) ON DELETE CASCADE,
    applicant_id INTEGER NOT NULL REFERENCES applicants(applicant_id) ON DELETE CASCADE,
    
    -- Donation details
    donation_amount DECIMAL(15,2) NOT NULL CHECK (donation_amount > 0),
    donation_request TEXT, -- Optional description or note from donor
    donation_status VARCHAR(20) DEFAULT 'PENDING' CHECK (donation_status IN ('PENDING', 'PROCESSING', 'COMPLETED', 'FAILED', 'REFUNDED', 'CANCELLED')),
    
    -- Funding progress update (calculated)
    funding_progress_after DECIMAL(15,2),
    
    -- Payment method
    payment_method VARCHAR(50),
    currency VARCHAR(3) DEFAULT 'BDT',
    
    -- Timestamps
    donation_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    processing_date TIMESTAMP,
    completion_date TIMESTAMP,
    
    -- Relationships
    transaction_id INTEGER UNIQUE, -- Will be set after transaction creation
    receipt_id INTEGER UNIQUE, -- Will be set after receipt generation
    
    -- Metadata
    is_anonymous BOOLEAN DEFAULT FALSE,
    donor_message TEXT,
    tags TEXT[],
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    -- Indexes
    INDEX idx_donation_status (donation_status),
    INDEX idx_donation_date (donation_date),
    INDEX idx_donor_applicant (donor_id, applicant_id),
    INDEX idx_donation_amount (donation_amount)
);

-- Transactions table
CREATE TABLE transactions (
    transaction_id SERIAL PRIMARY KEY,
    donation_id INTEGER UNIQUE NOT NULL REFERENCES donations(donation_id) ON DELETE CASCADE,
    
    -- Transaction details
    transaction_status VARCHAR(20) DEFAULT 'INITIATED' CHECK (transaction_status IN ('INITIATED', 'AUTHORIZED', 'PROCESSING', 'COMPLETED', 'FAILED', 'REFUNDED')),
    payment_gateway VARCHAR(50),
    gateway_transaction_id VARCHAR(100),
    gateway_response JSONB DEFAULT '{}'::jsonb,
    
    -- Amount details
    amount DECIMAL(15,2) NOT NULL,
    gateway_fee DECIMAL(10,2) DEFAULT 0.00,
    net_amount DECIMAL(15,2) NOT NULL,
    
    -- Security
    ip_address INET,
    user_agent TEXT,
    
    -- Timestamps
    initiated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    authorized_at TIMESTAMP,
    completed_at TIMESTAMP,
    failed_at TIMESTAMP,
    
    -- Error tracking
    error_code VARCHAR(50),
    error_message TEXT,
    retry_count INTEGER DEFAULT 0,
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    -- Indexes
    INDEX idx_transaction_status (transaction_status),
    INDEX idx_gateway_transaction (gateway_transaction_id),
    INDEX idx_transaction_dates (initiated_at, completed_at)
);

-- Receipts table
CREATE TABLE receipts (
    receipt_id SERIAL PRIMARY KEY,
    donation_id INTEGER UNIQUE NOT NULL REFERENCES donations(donation_id) ON DELETE CASCADE,
    
    -- Receipt details
    receipt_number VARCHAR(50) UNIQUE NOT NULL,
    confirmation_message TEXT,
    receipt_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    -- Content
    content JSONB DEFAULT '{}'::jsonb, -- Structured receipt data
    html_content TEXT, -- Formatted receipt for display
    pdf_url TEXT, -- Link to generated PDF receipt
    
    -- Tax information
    tax_deductible BOOLEAN DEFAULT FALSE,
    tax_certificate_url TEXT,
    
    -- Delivery
    sent_via_email BOOLEAN DEFAULT FALSE,
    email_sent_at TIMESTAMP,
    sent_via_sms BOOLEAN DEFAULT FALSE,
    sms_sent_at TIMESTAMP,
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    -- Indexes
    INDEX idx_receipt_number (receipt_number),
    INDEX idx_receipt_date (receipt_date)
);

-- ============================================
-- 4. NOTIFICATION AND REPORTING TABLES
-- ============================================

-- Notifications table
CREATE TABLE notifications (
    notification_id SERIAL PRIMARY KEY,
    
    -- Recipient information (polymorphic)
    recipient_type VARCHAR(20) NOT NULL CHECK (recipient_type IN ('APPLICANT', 'DONOR', 'VOLUNTEER', 'ADMINISTRATOR', 'ALL')),
    recipient_id INTEGER, -- Could be user_id, applicant_id, etc. based on type
    recipient_email VARCHAR(100),
    recipient_phone VARCHAR(20),
    
    -- Notification content
    notification_type VARCHAR(50) NOT NULL CHECK (notification_type IN (
        'DONATION_RECEIVED',
        'APPLICATION_STATUS_CHANGE',
        'EXPIRY_REMINDER',
        'VERIFICATION_UPDATE',
        'REPORT_GENERATED',
        'SYSTEM_ALERT',
        'PAYMENT_SUCCESS',
        'PAYMENT_FAILURE',
        'FUNDING_COMPLETE',
        'NEW_APPLICANT'
    )),
    title VARCHAR(200) NOT NULL,
    message TEXT NOT NULL,
    priority VARCHAR(10) DEFAULT 'MEDIUM' CHECK (priority IN ('LOW', 'MEDIUM', 'HIGH', 'URGENT')),
    
    -- Delivery status
    delivery_status VARCHAR(20) DEFAULT 'PENDING' CHECK (delivery_status IN ('PENDING', 'SENT', 'DELIVERED', 'READ', 'FAILED')),
    delivery_method VARCHAR(20) DEFAULT 'IN_APP' CHECK (delivery_method IN ('EMAIL', 'SMS', 'IN_APP', 'PUSH')),
    
    -- Timestamps
    scheduled_for TIMESTAMP,
    sent_at TIMESTAMP,
    delivered_at TIMESTAMP,
    read_at TIMESTAMP,
    
    -- Metadata
    metadata JSONB DEFAULT '{}'::jsonb,
    retry_count INTEGER DEFAULT 0,
    last_retry_at TIMESTAMP,
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    -- Indexes
    INDEX idx_notification_type (notification_type),
    INDEX idx_delivery_status (delivery_status),
    INDEX idx_recipient (recipient_type, recipient_id),
    INDEX idx_created_at (created_at)
);

-- Reports table
CREATE TABLE reports (
    report_id SERIAL PRIMARY KEY,
    generated_by INTEGER REFERENCES administrators(admin_id) ON DELETE SET NULL,
    
    -- Report details
    report_type VARCHAR(50) NOT NULL CHECK (report_type IN (
        'FINANCIAL_SUMMARY',
        'APPLICANT_STATS',
        'DONOR_STATS',
        'VOLUNTEER_PERFORMANCE',
        'SYSTEM_USAGE',
        'VERIFICATION_STATUS',
        'FUNDING_PROGRESS',
        'MONTHLY_REPORT',
        'QUARTERLY_REPORT',
        'ANNUAL_REPORT'
    )),
    report_title VARCHAR(200) NOT NULL,
    
    -- Content (store as JSON for flexibility)
    funding_statistics JSONB DEFAULT '{}'::jsonb,
    success_rates JSONB DEFAULT '{}'::jsonb,
    pending_verifications JSONB DEFAULT '{}'::jsonb,
    custom_metrics JSONB DEFAULT '{}'::jsonb,
    
    -- Date range
    period_start DATE NOT NULL,
    period_end DATE NOT NULL,
    
    -- Generation info
    generation_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    generation_duration INTERVAL,
    
    -- Export info
    export_format VARCHAR(20) DEFAULT 'JSON' CHECK (export_format IN ('JSON', 'PDF', 'CSV', 'EXCEL')),
    file_url TEXT,
    file_size BIGINT,
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    -- Constraints
    CONSTRAINT chk_period CHECK (period_end >= period_start),
    
    -- Indexes
    INDEX idx_report_type (report_type),
    INDEX idx_period (period_start, period_end),
    INDEX idx_generation_date (generation_date)
);

-- ============================================
-- 5. SYSTEM CONFIGURATION TABLE
-- ============================================

CREATE TABLE system_config (
    config_id SERIAL PRIMARY KEY,
    
    -- System components
    component VARCHAR(50) NOT NULL CHECK (component IN (
        'CENTRAL_DATABASE',
        'BACKEND_SERVER',
        'APPLICATION_SERVER',
        'STORAGE_SYSTEM',
        'INTERNET_CONNECTION',
        'PAYMENT_GATEWAY',
        'EMAIL_SERVICE',
        'SMS_SERVICE',
        'NOTIFICATION_SERVICE'
    )),
    
    -- Configuration details
    config_key VARCHAR(100) NOT NULL,
    config_value TEXT,
    config_type VARCHAR(20) DEFAULT 'STRING' CHECK (config_type IN ('STRING', 'NUMBER', 'BOOLEAN', 'JSON', 'ARRAY')),
    
    -- Status
    is_active BOOLEAN DEFAULT TRUE,
    last_verified_at TIMESTAMP,
    
    -- Metadata
    description TEXT,
    version VARCHAR(20),
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_by INTEGER REFERENCES administrators(admin_id) ON DELETE SET NULL,
    
    -- Indexes
    UNIQUE (component, config_key),
    INDEX idx_component (component),
    INDEX idx_is_active (is_active)
);

-- ============================================
-- 6. AUDIT LOG TABLE (For tracking all changes)
-- ============================================

CREATE TABLE audit_logs (
    log_id SERIAL PRIMARY KEY,
    
    -- Action details
    action_type VARCHAR(50) NOT NULL CHECK (action_type IN (
        'CREATE', 'UPDATE', 'DELETE', 'LOGIN', 'LOGOUT', 'VERIFY', 
        'APPROVE', 'REJECT', 'DONATE', 'TRANSACTION', 'REPORT_GENERATE'
    )),
    table_name VARCHAR(50) NOT NULL,
    record_id INTEGER NOT NULL,
    
    -- User who performed the action
    performed_by INTEGER REFERENCES users(user_id) ON DELETE SET NULL,
    performed_by_type VARCHAR(20),
    
    -- Changes
    old_values JSONB,
    new_values JSONB,
    changes JSONB,
    
    -- Context
    ip_address INET,
    user_agent TEXT,
    request_id VARCHAR(100),
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    -- Indexes
    INDEX idx_audit_table_record (table_name, record_id),
    INDEX idx_audit_user (performed_by),
    INDEX idx_audit_date (created_at)
);

-- ============================================
-- 7. TRIGGERS FOR AUTO-UPDATED TIMESTAMPS
-- ============================================

-- Function to update updated_at timestamp
CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = CURRENT_TIMESTAMP;
    RETURN NEW;
END;
$$ language 'plpgsql';

-- Create triggers for all tables with updated_at column
CREATE TRIGGER update_users_updated_at BEFORE UPDATE ON users
    FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

CREATE TRIGGER update_applicants_updated_at BEFORE UPDATE ON applicants
    FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

CREATE TRIGGER update_applicant_profiles_updated_at BEFORE UPDATE ON applicant_profiles
    FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

CREATE TRIGGER update_donations_updated_at BEFORE UPDATE ON donations
    FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

CREATE TRIGGER update_transactions_updated_at BEFORE UPDATE ON transactions
    FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

CREATE TRIGGER update_receipts_updated_at BEFORE UPDATE ON receipts
    FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

CREATE TRIGGER update_notifications_updated_at BEFORE UPDATE ON notifications
    FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

CREATE TRIGGER update_reports_updated_at BEFORE UPDATE ON reports
    FOR EACH ROW EXECUTE FUNCTION update_updated_at_column();

-- 8. VIEWS FOR COMMON QUERIES
-- ============================================

-- View for active volunteers with their applicants
CREATE VIEW active_volunteers_view AS
SELECT 
    v.volunteer_id,
    u.username,
    u.email,
    v.assigned_region,
    v.performance_score,
    v.total_applications_collected,
    COUNT(a.applicant_id) as active_applicants
FROM volunteers v
JOIN users u ON v.user_id = u.user_id
LEFT JOIN applicants a ON v.volunteer_id = a.volunteer_id 
    AND a.application_status IN ('UNDER_REVIEW', 'APPROVED', 'FUNDED')
WHERE u.account_status = 'ACTIVE'
GROUP BY v.volunteer_id, u.username, u.email, v.assigned_region, v.performance_score, v.total_applications_collected;

-- View for donor statistics
CREATE VIEW donor_statistics_view AS
SELECT 
    d.donor_id,
    u.username,
    u.email,
    d.donor_type,
    d.total_donated,
    d.donation_count,
    COUNT(dn.donation_id) as successful_donations,
    MIN(dn.donation_date) as first_donation,
    MAX(dn.donation_date) as last_donation
FROM donors d
JOIN users u ON d.user_id = u.user_id
LEFT JOIN donations dn ON d.donor_id = dn.donor_id AND dn.donation_status = 'COMPLETED'
WHERE u.account_status = 'ACTIVE'
GROUP BY d.donor_id, u.username, u.email, d.donor_type, d.total_donated, d.donation_count;

-- View for applicant funding progress
CREATE VIEW applicant_funding_view AS
SELECT 
    a.applicant_id,
    a.funding_goal,
    a.funding_progress,
    ROUND((a.funding_progress / a.funding_goal * 100), 2) as progress_percentage,
    a.application_status,
    a.submission_date,
    ap.upazilla,
    ap.union_name,
    COUNT(d.donation_id) as total_donations,
    SUM(d.donation_amount) as total_received
FROM applicants a
LEFT JOIN applicant_profiles ap ON a.applicant_id = ap.applicant_id
LEFT JOIN donations d ON a.applicant_id = d.applicant_id AND d.donation_status = 'COMPLETED'
GROUP BY a.applicant_id, a.funding_goal, a.funding_progress, a.application_status, 
         a.submission_date, ap.upazilla, ap.union_name;
